	tbl.Allowed = function(arg, arg2)
		local chat = arg.Chat
		if not chat then
			return true
		end

		if arg2 == chat.PinnedText or arg2 == chat.PinnedMeta or arg2 == chat.ReplyLabel then
			return false
		end

		if chat.Messages and arg2:IsDescendantOf(chat.Messages) then
			return arg2 == chat.EmptyState or arg2:IsA("TextButton")
		end
		return true
	end

	tbl.Track = function(arg, arg2)
		if arg.Bindings[arg2] or not arg:Allowed(arg2) then
			return
		end
		local str

		if arg2:IsA("TextBox") then
			str = "PlaceholderText"
		else
			if not (arg2:IsA("TextLabel") or arg2:IsA("TextButton")) then
				return
			end
			str = "Text"
		end

		local tbl3 = { Source = arg2[str], Property = str }
		arg.Bindings[arg2] = tbl3

		local function render()
			tbl3.Rendered = arg:Translate(tbl3.Source)

			if arg2[str] ~= tbl3.Rendered then
				arg2[str] = tbl3.Rendered
			end
		end

		tbl3.Render = render

		tbl3.Connection = arg2:GetPropertyChangedSignal(str):Connect(function()
			local v2 = arg2[str]
			if v2 == tbl3.Rendered then
				return
			end
			tbl3.Source = v2
			render()
		end)

		tbl3.Destroying = arg2.Destroying:Connect(function()
			tbl3.Connection:Disconnect()
			tbl3.Destroying:Disconnect()
			arg.Bindings[arg2] = nil
		end)

		render()
	end

	tbl.Set = function(arg, code)
		if code ~= "en" and code ~= "vi" then
			return
		end
		arg.Code = code
		arg:Save()

		for _, binding in pairs(arg.Bindings) do
			binding.Render()
		end
	end
